\makenoidxglossaries

\newglossaryentry{downscaling}{
    name=downscaling,
    description={The method of reducing distance between measurements in climate models by increasing the amount of data points}
}

\newglossaryentry{channel}{
    name=channel,
    description={A dimension in an image, which covers the entirety of the two spatial dimensions. In a regular image these are the "red", "green", and "blue" channels. In climate data these are often climatological variables like temperature or precipitation}
}

\newglossaryentry{scenario}{
    name=scenario,
    description={A carbon emission scenario. It gives the climate model a reference point for how much carbon will be present in the atmosphere at a given time. Read more \href{https://www.dkrz.de/en/communication/climate-simulations/cmip6-en/the-ssp-scenarios}{\url{https://www.dkrz.de/en/communication/climate-simulations/cmip6-en/the-ssp-scenarios}}. In this thesis we are working with SSP126 and SSP370}
}

\newglossaryentry{realisation}{
    name=realisation,
    description={One sample of an ensemble climate model. Realisations often vary in their initialisations, one can be initialised in a colder period and thus produce colder estimates of the climate and vice versa}
}

\newglossaryentry{ece}{
    name=EC-Earth,
    description={"European Consortium Earth 3" is a General Circulation Model used to create low-resolution climate models for the entire globe. In this thesis, EC-Earth-3 will be used as the low-resolution data for the downscaling task}
}

\newglossaryentry{hclim}{
    name=HCLIM,
    description={The Harmonie Climate Model is a Dynamical Downscaling Model used to create high-resolution climate models for the European domain. In this thesis HCLIM will be used as the high-resolution target data for the downscaling task.}
}

\newglossaryentry{126}{
    name=SSP1-2.6,
    description={Scenario 1 with 2.6 $w/m²$ additional radiative forcing. See \href{https://www.dkrz.de/en/communication/climate-simulations/cmip6-en/the-ssp-scenarios}{https://www.dkrz.de/en/communication/climate-simulations/cmip6-en/the-ssp-scenarios}}
}

\newglossaryentry{370}{
    name=SSP3-7.0,
    description={Scenario 3 with 7.0 $w/m²$ additional radiative forcing. See \href{https://www.dkrz.de/en/communication/climate-simulations/cmip6-en/the-ssp-scenarios}{https://www.dkrz.de/en/communication/climate-simulations/cmip6-en/the-ssp-scenarios}}
}

\newacronym
[description={Global climate models, that estimate the climate on the entire planet on a long temporal scale. Often in a 0.5° to 1.25\textdegree resolution}]
{gcm}{GCM}{General Circulation Model}


\newacronym
[description={Model created for downscaling the output General Circulation Models to a regional or local domain, usually improving specificity}]
{rcm}{RCM}{Regional Climate Model}

\newacronym
[description={Methods for \gls{downscaling}, in which data provides the general understanding (as opposed to physics). These methods are often Machine Learning based}]
{sdm}{SDM}{Statistical Downscaling Method}

\newacronym
[description={This is downscaling model similar to an SDM, but specifically this term covers physics-based downscaling models as opposed to the statistics-based SDM}]
{ddm}{DDM}{Dynamical Downscaling Model}

\newacronym
[description={The best of the best. Crème de la crème. The, until now, best method or model for a specific task}]
{sota}{SOTA}{State Of The Art}
